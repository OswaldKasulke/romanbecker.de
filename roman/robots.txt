# robots.txt — romanbecker.de
# Alle relevanten Crawler explizit erlaubt (inkl. AI-Bots).
# Ausnahme: die Strassenlinks der Stadtteilseiten.
#
# Die 94 Stadtteilseiten verlinken zusammen 5.956 Strassen auf das
# Bewertungsformular. Im HTML steht die Strasse als Fragment
# (/immobilienbewertung.html#strasse=...), Crawler rufen aber weiter die
# alte Query-Form ab, die sie aus ihrem Index kennen. Messung ueber die
# Access-Logs 28.08.-04.09.2026: 7.688 Abrufe mit strasse=, gegenueber
# 151 Abrufen der Bewertungsseite ohne Parameter — 43,7 % des gesamten
# Logvolumens. Verursacher: PetalBot 5.042, Bingbot 1.150, SemrushBot 274.
# Die Seite hat ein Self-Canonical auf die parameterlose URL, indexiert
# wurde also nichts; es ist reine Crawl-Last.
#
# Zwei Regeln je Block, weil ein Crawler nur seinen eigenen Block liest:
#   Disallow: /immobilienbewertung.html?   — reiner Praefix, auch fuer
#                                            einfache Parser eindeutig
#   Disallow: /*?strasse=                  — dieselben Parameter ueberall sonst
# Die parameterlose /immobilienbewertung.html bleibt erlaubt: sie ist
# kuerzer als die Regel und wird vom Praefix nicht getroffen.
# gclid/gbraid/wbraid bleiben bewusst offen (Ads-Zielseiten).

User-agent: *
Allow: /
Disallow: /immobilienbewertung.html?
Disallow: /*?strasse=
